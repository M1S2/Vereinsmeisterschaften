91 public async Task<bool> Load(string path, CancellationToken cancellationToken)
92 {
93 bool importingResult = false;
94 Exception exception = null;
95 await Task.Run(() =>
96 {
97 try
98 {
99 if (!File.Exists(path))
100 {
101 OnFileProgress?.Invoke(this, 0);
102 ClearAll();
103 OnFileProgress?.Invoke(this, 100);
104 }
105 else
106 {
107 List<WinnerHistoryElement> list = _fileService.LoadFromCsv<WinnerHistoryElement>(path, cancellationToken, WinnerHistoryElement.SetPropertyFromString, out _, OnFileProgress, (header) =>
108 {
109 return PropertyNameLocalizedStringHelper.FindProperty(typeof(WinnerHistoryElement), header);
110 });
111 _previousWinnersList = new ObservableCollection<WinnerHistoryElement>();
112 foreach (WinnerHistoryElement previousWinner in list)
113 {
114 addPreviousWinnerInternal(previousWinner, false);
115 }
116 }
117
118 _previousWinnersListOnLoad = _previousWinnersList.ToList().ConvertAll(p => new WinnerHistoryElement(p));
119
120 sortPreviousWinners();
121
122 PersistentPath = path;
123 importingResult = true;
124 }
125 catch (OperationCanceledException)
126 {
127 importingResult = false;
128 }
129 catch (Exception ex)
130 {
131 exception = ex;
132 }
133 });
134 OnFileFinished?.Invoke(this, null);
135 if (exception != null) { throw exception; }
136 return importingResult;
137 }

147 public async Task<bool> Save(CancellationToken cancellationToken, string path = "")
148 {
149 if (string.IsNullOrEmpty(path)) { path = PersistentPath; }
150
151 bool saveResult = false;
152 Exception exception = null;
153 await Task.Run(() =>
154 {
155 try
156 {
157 _fileService.SaveToCsv(path, _previousWinnersList.ToList(), cancellationToken, null, OnFileProgress, (data, parentObject, currentProperty) =>
158 {
159 if (data is IList dataList)
160 {
161 return string.Join(" | ", dataList.Cast<object>().Select(d => d.ToString()));
162 }
163 else if(data is DateTime dataDate)
164 {
165 return dataDate.ToShortDateString();
166 }
167 else
168 {
169 return data?.ToString();
170 }
171 },
172 (header, type) =>
173 {
174 return PropertyNameLocalizedStringHelper.Convert(typeof(WinnerHistoryElement), header);
175 });
176
177 _previousWinnersListOnLoad = _previousWinnersList.ToList().ConvertAll(p => new WinnerHistoryElement(p));
178 saveResult = true;
179 }
180 catch (OperationCanceledException)
181 {
182 saveResult = false;
183 }
184 catch (Exception ex)
185 {
186 exception = ex;
187 }
188 });
189 OnFileFinished?.Invoke(this, null);
190 if (exception != null) { throw exception; }
191 return saveResult;

322 public void UpdatePreviousWinnersHistoryFlags()
323 {
324 if (_previousWinnersList == null || _previousWinnersList.Count == 0) { return; }
325
326 // Collect the names of all current winners (from all podium places)
327 List<WinnerName> currentWinnersCombined = new[] { CurrentWinners?.WinnersGold, CurrentWinners?.WinnersSilver, CurrentWinners?.WinnersBronze }
328 .Where(list => list != null)
329 .SelectMany(list => list)
330 .ToList();
331
332 foreach (WinnerHistoryElement historyElement in _previousWinnersList)
333 {
334 // Check if the winner history element is the current one.
335 historyElement.MatchesCurrentWinnersElement = historyElement.Equals(CurrentWinners);
336
337 // Check for each podium place of the winner history element if the current winner names are contained.
338 historyElement.WinnersGold?.ForEach(name => name.IsPartOfCurrentWinners = currentWinnersCombined.Contains(name));
339 historyElement.WinnersSilver?.ForEach(name => name.IsPartOfCurrentWinners = currentWinnersCombined.Contains(name));
340 historyElement.WinnersBronze?.ForEach(name => name.IsPartOfCurrentWinners = currentWinnersCombined.Contains(name));
341 }
342 }

347 public Dictionary<WinnerName, int> GetPreviousWinnersCountPerPerson(ResultPodiumsPlaces podiumPlace = ResultPodiumsPlaces.All, bool ignoreCurrentWinners = false)
348 {
349 IEnumerable<WinnerHistoryElement> previousWinners = ignoreCurrentWinners ? _previousWinnersList.Except(new List<WinnerHistoryElement>() { CurrentWinners }).Where(p => !p.MatchesCurrentWinnersElement) : _previousWinnersList;
350
351 List<WinnerName> podiumWinnerNames = new List<WinnerName>();
352 switch (podiumPlace)
353 {
354 case ResultPodiumsPlaces.Gold: podiumWinnerNames = previousWinners.Where(h => h.WinnersGold != null).SelectMany(h => h.WinnersGold).ToList(); break;
355 case ResultPodiumsPlaces.Silver: podiumWinnerNames = previousWinners.Where(h => h.WinnersSilver != null).SelectMany(h => h.WinnersSilver).ToList(); break;
356 case ResultPodiumsPlaces.Bronze: podiumWinnerNames = previousWinners.Where(h => h.WinnersBronze != null).SelectMany(h => h.WinnersBronze).ToList(); break;
357 case ResultPodiumsPlaces.All:
358 podiumWinnerNames = previousWinners.SelectMany(h => (h.WinnersGold ?? Enumerable.Empty<WinnerName>())
359 .Concat(h.WinnersSilver ?? Enumerable.Empty<WinnerName>())
360 .Concat(h.WinnersBronze ?? Enumerable.Empty<WinnerName>())).ToList();
361 break;
362 default: break;
363 }
364
365 Dictionary<WinnerName, int> podiumCounts = podiumWinnerNames.GroupBy(winner => winner)
366 .OrderByDescending(g => g.Count())
367 .ToDictionary(g => g.Key, g => g.Count());
368 return podiumCounts;
369 }
